### 3.1 坑的形状变了，别说错

```python
import os, sys, glob, re, subprocess
root = sys.argv[1] if len(sys.argv) > 1 else '.'
def rc(args): return subprocess.run(args, cwd=root, capture_output=True, text=True).returncode
def lic_of(p):
    t=open(p,encoding='utf-8',errors='replace').read()
    m=re.search(r'^license:\s*(.+)$',t,re.M)
    return m.group(1).strip().strip('"') if m else ''
red=[]; pending=[]; n=0
for p in sorted(glob.glob(os.path.join(root,'*/*/SKILL.md'))):
    rel=os.path.relpath(p,root); top=rel.split(os.sep)[0]
    if top not in ('skill','skills','agent-skills'): continue
    n+=1; lic=lic_of(p)
    ignored=rc(['git','check-ignore','-q',rel])==0
    tracked=rc(['git','ls-files','--error-unmatch',rel])==0
    thirdparty=('MIT' in lic) or ('未声明' in lic)
    if top=='agent-skills':
        if ignored: red.append((rel,'agent-skills 被忽略'))
        elif not tracked: red.append((rel,'agent-skills 未入库'))
    elif top=='skills':   # 第三方纳管目录：只允许第三方
        if ignored and not thirdparty: red.append((rel,'自有 license 却在 skills/ 被忽略目录'))
        elif not ignored and not tracked: red.append((rel,'skills/ 里既未忽略也未跟踪'))
    else:                 # skill/：提取件目录，必须显式忽略，不许滑成第三种状态
        if not ignored: red.append((rel,'skill/ 提取件未被忽略（可被 git add -A 带走）'))
# 提取目录的两份来历文档必须未被忽略；未跟踪是待办，不是危险态
for doc in ('skill/README.md','skill/标准化台账-20261017.md'):
    p=os.path.join(root,doc)
    if not os.path.exists(p): continue
    n+=1
    if rc(['git','check-ignore','-q',doc])==0:
        red.append((doc,'来历文档被忽略，来路说明丢失'))
    elif rc(['git','ls-files','--error-unmatch',doc])!=0:
        pending.append(doc)
print('rows=%d red=%d pending=%d' % (n,len(red),len(pending)))
for rel,why in red: print('RED',rel,'::',why)
for rel in pending: print('PENDING',rel,':: 已放行但未入库（待 git add）')
sys.exit(1 if red else 0)
```
